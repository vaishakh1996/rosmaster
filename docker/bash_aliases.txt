
# ROS 2 Humble - CPU-only development aliases

# ROS workspace
export ROS_WS="$HOME/ros2_ws"

# Build workspace explicitly using the existing script
alias build='bash "$HOME/workspace.sh"'

# Navigate to workspace
alias ws='cd "$ROS_WS"'
alias src='cd "$ROS_WS/src"'

# Source ROS Humble and the workspace overlay
alias rosreload='source /opt/ros/humble/setup.bash; if [ -f "$ROS_WS/install/setup.bash" ]; then source "$ROS_WS/install/setup.bash"; fi'

# ROS inspection
alias ros2top='ros2 topic list'
alias ros2nodes='ros2 node list'
alias ros2services='ros2 service list'
alias ros2params='ros2 param list'

# ROS demonstration nodes
alias talker='ros2 run demo_nodes_cpp talker'
alias listener='ros2 run demo_nodes_py listener'

# Visualization (CPU software rendering)
alias rviz='LIBGL_ALWAYS_SOFTWARE=1 rviz2'

# Display ROS environment variables
alias rosenv='env | grep -E "ROS|AMENT|COLCON|XDG" | sort'

# Reload interactive shell configuration
alias reload='source ~/.bashrc'

# Show processes without terminating them
alias rosps='ps aux | grep -E "[r]os2|[r]viz2|[g]azebo|[g]z sim"'
